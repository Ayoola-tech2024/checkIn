   321→   321→      resizeCanvas.height = canvas.height * scale;

   324→   324→        ctx.drawImage(canvas, 0, 0, resizeCanvas.width, resizeCanvas.height);

   326→   326→      }
   327→   327→    }

   332→   332→  // ─── Generate deterministic descriptor from image data ────
   333→   333→  const generateImageHashDescriptor = useCallback((canvas: HTMLCanvasElement, faceBox?: { x: number; y: number; width: number; height: number }): number[] => {
   334→   334→    const thumbW = 8;
   335→   335→    const thumbH = 16;
   336→   336→    const thumbCanvas = document.createElement('canvas');
   337→   337→    thumbCanvas.width = thumbW;
   338→   338→    thumbCanvas.height = thumbH;
   339→   339→    const thumbCtx = thumbCanvas.getContext('2d');
   340→   340→
   341→   341→    if (!thumbCtx) return Array.from({ length: 128 }, () => 0);
   342→   342→
   343→   343→    let sx: number, sy: number, sw: number, sh: number;
   344→   344→    if (faceBox) {
   345→   345→      const pad = Math.max(faceBox.width, faceBox.height) * 0.2;
   346→   346→      sx = Math.max(0, faceBox.x - pad);
   347→   347→      sy = Math.max(0, faceBox.y - pad);
   348→   348→      sw = Math.min(canvas.width - sx, faceBox.width + pad * 2);
   349→   349→      sh = Math.min(canvas.height - sy, faceBox.height + pad * 2);
   350→   350→    } else {
   351→   351→      sx = Math.floor(canvas.width * 0.25);
   352→   352→      sy = Math.floor(canvas.height * 0.1);
   353→   353→      sw = Math.floor(canvas.width * 0.5);
   354→   354→      sh = Math.floor(canvas.height * 0.65);
   355→   355→    }
   356→   356→
   357→   357→    thumbCtx.drawImage(canvas, sx, sy, sw, sh, 0, 0, thumbW, thumbH);
   358→   358→
   359→   359→    const imageData = thumbCtx.getImageData(0, 0, thumbW, thumbH);
   360→   360→    const pixels = imageData.data;
   361→   361→
   362→   362→    const descriptor: number[] = [];
   363→   363→    for (let i = 0; i < 128; i++) {
   364→   364→      const offset = i * 4;
   365→   365→      if (offset + 2 < pixels.length) {
   366→   366→        const r = pixels[offset] / 255;
   367→   367→        const g = pixels[offset + 1] / 255;
   368→   368→        const b = pixels[offset + 2] / 255;
   369→   369→        const luminance = 0.299 * r + 0.587 * g + 0.114 * b;
   370→   370→        descriptor.push(luminance * 2 - 1);
   371→   371→      } else {
   372→   372→        descriptor.push(0);
   373→   373→      }
   374→   374→    }
   375→   375→
   376→   376→    const transformed: number[] = [];
   377→   377→    for (let k = 0; k < 128; k++) {
   378→   378→      let sum = 0;
   379→   379→      for (let n = 0; n < 128; n++) {
   380→   380→        sum += descriptor[n] * Math.cos((Math.PI * k * (n + 0.5)) / 128);
   381→   381→      }
   382→   382→      transformed.push(sum / 128);
   383→   383→    }
   384→   384→
   385→   385→    const magnitude = Math.sqrt(transformed.reduce((sum, v) => sum + v * v, 0));
   386→   386→    if (magnitude > 0) {
   387→   387→      for (let i = 0; i < transformed.length; i++) {
   388→   388→        transformed[i] = transformed[i] / magnitude;
   389→   389→      }
   390→   390→    }
   391→   391→
   392→   392→    return transformed;
   393→   393→  }, []);

   395→   395→  // ─── Process an image (from camera or upload) through face-api ───
   396→   396→  const processImage = useCallback(async (imageSource: HTMLCanvasElement | HTMLImageElement): Promise<{ selfieData: string; facialDescriptor: number[] } | null> => {
   397→   397→    const canvas = document.createElement('canvas');
   398→   398→    const width = imageSource instanceof HTMLVideoElement ? (imageSource.videoWidth || 640) :
   399→   399→                  imageSource instanceof HTMLCanvasElement ? imageSource.width :
   400→   400→                  imageSource.naturalWidth || imageSource.width;
   401→   401→    const height = imageSource instanceof HTMLVideoElement ? (imageSource.videoHeight || 480) :
   402→   402→                   imageSource instanceof HTMLCanvasElement ? imageSource.height :
   403→   403→                   imageSource.naturalHeight || imageSource.height;
   404→   404→    canvas.width = width;
   405→   405→    canvas.height = height;
   406→   406→    const ctx = canvas.getContext('2d');
   407→   407→    if (!ctx) return null;
   408→   408→
   409→   409→    // Draw image to canvas
   410→   410→    if (imageSource instanceof HTMLVideoElement) {
   411→   411→      // Mirror for selfie
   412→   412→      ctx.save();
   413→   413→      ctx.translate(canvas.width, 0);
   414→   414→      ctx.scale(-1, 1);
   415→   415→      ctx.drawImage(imageSource, 0, 0, canvas.width, canvas.height);
   416→   416→      ctx.restore();
   417→   417→    } else {
   418→   418→      ctx.drawImage(imageSource, 0, 0, canvas.width, canvas.height);
   419→   419→    }
   420→   420→
   421→   421→    const selfieData = compressImage(canvas);
   422→   422→
   423→   423→    // Try face-api.js detection
   424→   424→    let detectedFaceBox: { x: number; y: number; width: number; height: number } | undefined;
   425→   425→
   426→   426→    if (faceApiLoaded && faceApiModule) {
   427→   427→      try {
   428→   428→        console.log('[FaceCapture] Running face detection on image...');
   429→   429→        const faceapi = faceApiModule;
   430→   430→
   431→   431→        let fullDetection;
   432→   432→        if (faceapi.nets.tinyFaceDetector.isLoaded && faceapi.nets.faceRecognitionNet.isLoaded) {
   433→   433→          try {
   434→   434→            fullDetection = await faceapi
   435→   435→              .detectSingleFace(canvas, new faceapi.TinyFaceDetectorOptions({ inputSize: 224, scoreThreshold: 0.2 }))
   436→   436→              .withFaceLandmarks()
   437→   437→              .withFaceDescriptor();
   438→   438→          } catch (e) {
   439→   439→            console.warn('[FaceCapture] TinyFaceDetector + descriptor failed:', e);
   440→   440→          }
   441→   441→        }
   442→   442→
   443→   443→        if (!fullDetection && faceapi.nets.ssdMobilenetv1.isLoaded && faceapi.nets.faceRecognitionNet.isLoaded) {
   444→   444→          try {
   445→   445→            fullDetection = await faceapi
   446→   446→              .detectSingleFace(canvas, new faceapi.SsdMobilenetv1Options({ minConfidence: 0.2 }))
   447→   447→              .withFaceLandmarks()
   448→   448→              .withFaceDescriptor();
   449→   449→          } catch (e) {
   450→   450→            console.warn('[FaceCapture] SsdMobilenetv1 + descriptor failed:', e);
   451→   451→          }
   452→   452→        }
   453→   453→
   454→   454→        if (fullDetection) {
   455→   455→          const descriptor = Array.from(fullDetection.descriptor) as number[];
   456→   456→          console.log('[FaceCapture] Face detected with descriptor! Length:', descriptor.length);
   457→   457→          return { selfieData, facialDescriptor: descriptor };
   458→   458→        }
   459→   459→
   460→   460→        // Try detection only for face box
   461→   461→        let simpleDetection;
   462→   462→        if (faceapi.nets.tinyFaceDetector.isLoaded) {
   463→   463→          try {
   464→   464→            simpleDetection = await faceapi
   465→   465→              .detectSingleFace(canvas, new faceapi.TinyFaceDetectorOptions({ inputSize: 224, scoreThreshold: 0.15 }));
   466→   466→          } catch (e) {
   467→   467→            console.warn('[FaceCapture] Simple TinyFace detection failed:', e);
   468→   468→          }
