   307→  // ─── Compress image to under MAX_SELFIE_SIZE_KB ──────────
   308→  const compressImage = useCallback((canvas: HTMLCanvasElement): string => {
   309→    let quality = 0.8;
   310→    let dataUrl = canvas.toDataURL('image/jpeg', quality);
   311→
   312→    while (dataUrl.length * 0.75 > MAX_SELFIE_SIZE_KB * 1024 && quality > 0.1) {
   313→      quality -= 0.1;
   314→      dataUrl = canvas.toDataURL('image/jpeg', quality);
   315→    }
   316→
   317→    if (dataUrl.length * 0.75 > MAX_SELFIE_SIZE_KB * 1024) {
   318→      const scale = 0.75;
   319→      const resizeCanvas = document.createElement('canvas');
   320→      resizeCanvas.width = canvas.width * scale;
   321→      resizeCanvas.height = canvas.height * scale;
   322→      const ctx = resizeCanvas.getContext('2d');
   323→      if (ctx) {
   324→        ctx.drawImage(canvas, 0, 0, resizeCanvas.width, resizeCanvas.height);
   325→        dataUrl = resizeCanvas.toDataURL('image/jpeg', 0.7);
   326→      }
   327→    }
   328→
   329→    return dataUrl;
   330→  }, []);

   332→  // ─── Generate deterministic descriptor from image data ────
   333→  const generateImageHashDescriptor = useCallback((canvas: HTMLCanvasElement, faceBox?: { x: number; y: number; width: number; height: number }): number[] => {
   334→    const thumbW = 8;
   335→    const thumbH = 16;
   336→    const thumbCanvas = document.createElement('canvas');
   337→    thumbCanvas.width = thumbW;
   338→    thumbCanvas.height = thumbH;
   339→    const thumbCtx = thumbCanvas.getContext('2d');
   340→
   341→    if (!thumbCtx) return Array.from({ length: 128 }, () => 0);
   342→
   343→    let sx: number, sy: number, sw: number, sh: number;
   344→    if (faceBox) {
   345→      const pad = Math.max(faceBox.width, faceBox.height) * 0.2;
   346→      sx = Math.max(0, faceBox.x - pad);
   347→      sy = Math.max(0, faceBox.y - pad);
   348→      sw = Math.min(canvas.width - sx, faceBox.width + pad * 2);
   349→      sh = Math.min(canvas.height - sy, faceBox.height + pad * 2);
   350→    } else {
   351→      sx = Math.floor(canvas.width * 0.25);
   352→      sy = Math.floor(canvas.height * 0.1);
   353→      sw = Math.floor(canvas.width * 0.5);
   354→      sh = Math.floor(canvas.height * 0.65);
   355→    }
   356→
   357→    thumbCtx.drawImage(canvas, sx, sy, sw, sh, 0, 0, thumbW, thumbH);
   358→
   359→    const imageData = thumbCtx.getImageData(0, 0, thumbW, thumbH);
   360→    const pixels = imageData.data;
   361→
   362→    const descriptor: number[] = [];
   363→    for (let i = 0; i < 128; i++) {
   364→      const offset = i * 4;
   365→      if (offset + 2 < pixels.length) {
   366→        const r = pixels[offset] / 255;
   367→        const g = pixels[offset + 1] / 255;
   368→        const b = pixels[offset + 2] / 255;
   369→        const luminance = 0.299 * r + 0.587 * g + 0.114 * b;
   370→        descriptor.push(luminance * 2 - 1);
   371→      } else {
   372→        descriptor.push(0);
   373→      }
   374→    }
   375→
   376→    const transformed: number[] = [];
   377→    for (let k = 0; k < 128; k++) {
   378→      let sum = 0;
   379→      for (let n = 0; n < 128; n++) {
   380→        sum += descriptor[n] * Math.cos((Math.PI * k * (n + 0.5)) / 128);
   381→      }
   382→      transformed.push(sum / 128);
   383→    }
   384→
   385→    const magnitude = Math.sqrt(transformed.reduce((sum, v) => sum + v * v, 0));
   386→    if (magnitude > 0) {
   387→      for (let i = 0; i < transformed.length; i++) {
   388→        transformed[i] = transformed[i] / magnitude;
   389→      }
   390→    }
   391→
   392→    return transformed;
   393→  }, []);

   395→  // ─── Process an image (from camera or upload) through face-api ───
   396→  const processImage = useCallback(async (imageSource: HTMLCanvasElement | HTMLImageElement): Promise<{ selfieData: string; facialDescriptor: number[] } | null> => {
   397→    const canvas = document.createElement('canvas');
   398→    const width = imageSource instanceof HTMLVideoElement ? (imageSource.videoWidth || 640) :
   399→                  imageSource instanceof HTMLCanvasElement ? imageSource.width :
   400→                  imageSource.naturalWidth || imageSource.width;
   401→    const height = imageSource instanceof HTMLVideoElement ? (imageSource.videoHeight || 480) :
   402→                   imageSource instanceof HTMLCanvasElement ? imageSource.height :
   403→                   imageSource.naturalHeight || imageSource.height;
   404→    canvas.width = width;
   405→    canvas.height = height;
   406→    const ctx = canvas.getContext('2d');
   407→    if (!ctx) return null;
   408→
   409→    // Draw image to canvas
   410→    if (imageSource instanceof HTMLVideoElement) {
   411→      // Mirror for selfie
   412→      ctx.save();
   413→      ctx.translate(canvas.width, 0);
   414→      ctx.scale(-1, 1);
   415→      ctx.drawImage(imageSource, 0, 0, canvas.width, canvas.height);
   416→      ctx.restore();
   417→    } else {
   418→      ctx.drawImage(imageSource, 0, 0, canvas.width, canvas.height);
   419→    }
   420→
   421→    const selfieData = compressImage(canvas);
   422→
   423→    // Try face-api.js detection
   424→    let detectedFaceBox: { x: number; y: number; width: number; height: number } | undefined;
   425→
   426→    if (faceApiLoaded && faceApiModule) {
   427→      try {
   428→        console.log('[FaceCapture] Running face detection on image...');
   429→        const faceapi = faceApiModule;
   430→
   431→        let fullDetection;
   432→        if (faceapi.nets.tinyFaceDetector.isLoaded && faceapi.nets.faceRecognitionNet.isLoaded) {
   433→          try {
   434→            fullDetection = await faceapi
   435→              .detectSingleFace(canvas, new faceapi.TinyFaceDetectorOptions({ inputSize: 224, scoreThreshold: 0.2 }))
   436→              .withFaceLandmarks()
   437→              .withFaceDescriptor();
   438→          } catch (e) {
   439→            console.warn('[FaceCapture] TinyFaceDetector + descriptor failed:', e);
   440→          }
   441→        }
   442→
   443→        if (!fullDetection && faceapi.nets.ssdMobilenetv1.isLoaded && faceapi.nets.faceRecognitionNet.isLoaded) {
   444→          try {
   445→            fullDetection = await faceapi
   446→              .detectSingleFace(canvas, new faceapi.SsdMobilenetv1Options({ minConfidence: 0.2 }))
   447→              .withFaceLandmarks()
   448→              .withFaceDescriptor();
   449→          } catch (e) {
   450→            console.warn('[FaceCapture] SsdMobilenetv1 + descriptor failed:', e);
   451→          }
   452→        }
   453→
   454→        if (fullDetection) {
   455→          const descriptor = Array.from(fullDetection.descriptor) as number[];
   456→          console.log('[FaceCapture] Face detected with descriptor! Length:', descriptor.length);
   457→          return { selfieData, facialDescriptor: descriptor };
   458→        }
   459→
   460→        // Try detection only for face box
   461→        let simpleDetection;
   462→        if (faceapi.nets.tinyFaceDetector.isLoaded) {
   463→          try {
   464→            simpleDetection = await faceapi
   465→              .detectSingleFace(canvas, new faceapi.TinyFaceDetectorOptions({ inputSize: 224, scoreThreshold: 0.15 }));
   466→          } catch (e) {
   467→            console.warn('[FaceCapture] Simple TinyFace detection failed:', e);
   468→          }
   469→        }
   470→        if (!simpleDetection && faceapi.nets.ssdMobilenetv1.isLoaded) {
   471→          try {
   472→            simpleDetection = await faceapi
   473→              .detectSingleFace(canvas, new faceapi.SsdMobilenetv1Options({ minConfidence: 0.15 }));
   474→          } catch (e) {
   475→            console.warn('[FaceCapture] Simple SSD detection failed:', e);
   476→          }
   477→        }
   478→
   479→        if (simpleDetection) {
   480→          const box = simpleDetection.box;
   481→          detectedFaceBox = { x: box.x, y: box.y, width: box.width, height: box.height };
   482→          console.log('[FaceCapture] Face box detected (no descriptor). Using image hash with face crop.');
   483→        } else {
   484→          console.warn('[FaceCapture] No face detected. Using image hash with center crop.');
   485→        }
   486→      } catch (detectErr) {
   487→        console.error('[FaceCapture] Face detection error:', detectErr);
   488→      }
   489→    }
   490→
   491→    // Fallback: image-hash-based descriptor
   492→    console.log('[FaceCapture] Using image-hash descriptor', detectedFaceBox ? '(with face crop)' : '(center crop)');
   493→    const descriptor = generateImageHashDescriptor(canvas, detectedFaceBox);
   494→    return { selfieData, facialDescriptor: descriptor };
   495→  }, [compressImage, faceApiLoaded, generateImageHashDescriptor]);

   497→  // ─── Handle camera capture ──────────────────────────────────
   498→  const handleCameraCapture = useCallback(async () => {

   514→        onError('Failed to capture image - canvas context error');

   516→        return;
   517→      }
   518→
   519→      // Mirror the image for selfie
   520→      ctx.save();
   521→      ctx.translate(canvas.width, 0);
   522→      ctx.scale(-1, 1);
   523→      ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
   524→      ctx.restore();
   525→
   526→      const selfieData = compressImage(canvas);
   527→      setCapturedImage(selfieData);
   528→
   529→      // Process through face-api
   530→      let detectedFaceBox: { x: number; y: number; width: number; height: number } | undefined;
   531→
   532→      if (faceApiLoaded && faceApiModule) {
   533→        try {
   534→          console.log('[FaceCapture] Running face detection on captured image...');
   535→          const faceapi = faceApiModule;
   536→
   537→          let fullDetection;
   538→          if (faceapi.nets.tinyFaceDetector.isLoaded && faceapi.nets.faceRecognitionNet.isLoaded) {
   539→            try {
   540→              fullDetection = await faceapi
   541→                .detectSingleFace(canvas, new faceapi.TinyFaceDetectorOptions({ inputSize: 224, scoreThreshold: 0.2 }))
   542→                .withFaceLandmarks()
   543→                .withFaceDescriptor();
   544→            } catch (e) {
   545→              console.warn('[FaceCapture] TinyFaceDetector + landmarks + descriptor failed:', e);
   546→            }
   547→          }
   548→
   549→          if (!fullDetection && faceapi.nets.ssdMobilenetv1.isLoaded && faceapi.nets.faceRecognitionNet.isLoaded) {
   550→            try {
   551→              fullDetection = await faceapi
   552→                .detectSingleFace(canvas, new faceapi.SsdMobilenetv1Options({ minConfidence: 0.2 }))
   553→                .withFaceLandmarks()
   554→                .withFaceDescriptor();
   555→            } catch (e) {
   556→              console.warn('[FaceCapture] SsdMobilenetv1 + landmarks + descriptor failed:', e);
   557→            }
   558→          }
   559→
   560→          if (fullDetection) {
   561→            const descriptor = Array.from(fullDetection.descriptor) as number[];
   562→            console.log('[FaceCapture] Face detected with descriptor! Length:', descriptor.length);
   563→            setDetectionStatus('face-found');
   564→            onCapture({ selfieData, facialDescriptor: descriptor });
   565→            setCapturing(false);
   566→            return;
   567→          }
   568→
   569→          // Try face detection only (without descriptor)
   570→          let simpleDetection;
   571→          if (faceapi.nets.tinyFaceDetector.isLoaded) {
   572→            try {
   573→              simpleDetection = await faceapi
   574→                .detectSingleFace(canvas, new faceapi.TinyFaceDetectorOptions({ inputSize: 224, scoreThreshold: 0.1 }));
   575→            } catch (e) {
   576→              console.warn('[FaceCapture] Simple detection failed:', e);
   577→            }
   578→          }
   579→          if (!simpleDetection && faceapi.nets.ssdMobilenetv1.isLoaded) {
   580→            try {
   581→              simpleDetection = await faceapi
   582→                .detectSingleFace(canvas, new faceapi.SsdMobilenetv1Options({ minConfidence: 0.1 }));
   583→            } catch (e) {
   584→              console.warn('[FaceCapture] SSD simple detection failed:', e);
   585→            }
   586→          }
   587→
   588→          if (simpleDetection) {
   589→            const box = simpleDetection.box;
   590→            detectedFaceBox = { x: box.x, y: box.y, width: box.width, height: box.height };
   591→            setDetectionStatus('face-found');
   592→          }
   593→        } catch (detectErr) {
   594→          console.error('[FaceCapture] Face detection error on capture:', detectErr);
   595→        }
   596→      }
   597→
   598→      // Fallback: image-hash-based descriptor (always works, even without face detection)
   599→      // DEMO: This always produces a valid descriptor so the capture always succeeds
   600→      console.log('[FaceCapture] Using image-hash descriptor', detectedFaceBox ? '(with face crop)' : '(center crop - demo mode)');
   601→      const descriptor = generateImageHashDescriptor(canvas, detectedFaceBox);
   602→      setDetectionStatus('face-found');
   603→      onCapture({ selfieData, facialDescriptor: descriptor });
   604→    } catch (err) {
   605→      console.error('[FaceCapture] Capture error:', err);
   606→      onError('Failed to process capture. Please try again.');
   607→    } finally {
   608→      setCapturing(false);
   609→    }
