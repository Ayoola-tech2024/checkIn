    99→function SessionStatusBadge({ status }: { status: string }) {
   100→  const config: Record<string, { variant: 'default' | 'secondary' | 'destructive' | 'outline'; className: string; label: string }> = {
   101→    scheduled: { variant: 'outline', className: 'border-slate-400 text-slate-600 dark:text-slate-300', label: 'Scheduled' },
   102→    active: { variant: 'default', className: 'bg-emerald-600 hover:bg-emerald-700 text-white animate-pulse', label: 'Active' },
   103→    completed: { variant: 'secondary', className: '', label: 'Completed' },
   104→    cancelled: { variant: 'destructive', className: '', label: 'Cancelled' },
   105→  };
   106→  const c = config[status] || { variant: 'outline' as const, className: '', label: status };
   107→  return (
   108→    <Badge variant={c.variant} className={c.className}>
   109→      {c.label}
   110→    </Badge>
   111→  );
   112→}

   141→function CreateSessionDialog({ open, onOpenChange, lecturerId, courses, onCreated }: CreateSessionDialogProps) {
   142→  const [title, setTitle] = useState('');
   143→  const [courseId, setCourseId] = useState('');
   144→  const [venueId, setVenueId] = useState('');
   145→  const [level, setLevel] = useState<string>('100');
   146→  const [selectedDepts, setSelectedDepts] = useState<string[]>([]);
   147→  const [distanceThreshold, setDistanceThreshold] = useState(DEFAULT_DISTANCE_THRESHOLD);
   148→  const [durationMinutes, setDurationMinutes] = useState(DEFAULT_SESSION_DURATION);
   149→  const [scheduledAt, setScheduledAt] = useState('');
   150→  const [venues, setVenues] = useState<VenueInfo[]>([]);
   151→  const [departments, setDepartments] = useState<DepartmentInfo[]>([]);
   152→  const [saving, setSaving] = useState(false);
   153→
   154→  // Fetch venues and departments when dialog opens
   155→  useEffect(() => {
   156→    if (open) {
   157→      fetch('/api/admin/venues')
   158→        .then((r) => r.json())
   159→        .then((json) => { if (json.success) setVenues(json.data); })
   160→        .catch(() => toast.error('Failed to load venues'));
   161→
   162→      fetch('/api/admin/departments')
   163→        .then((r) => r.json())
   164→        .then((json) => { if (json.success) setDepartments(json.data); })
   165→        .catch(() => toast.error('Failed to load departments'));
   166→    }
   167→  }, [open]);
   168→
   169→  const resetForm = () => {
   170→    setTitle('');
   171→    setCourseId('');
   172→    setVenueId('');
   173→    setLevel('100');
   174→    setSelectedDepts([]);
   175→    setDistanceThreshold(DEFAULT_DISTANCE_THRESHOLD);
   176→    setDurationMinutes(DEFAULT_SESSION_DURATION);
   177→    setScheduledAt('');
   178→  };
   179→
   180→  const handleDeptToggle = (deptId: string) => {
   181→    setSelectedDepts((prev) =>
   182→      prev.includes(deptId) ? prev.filter((d) => d !== deptId) : [...prev, deptId]
   183→    );
   184→  };
   185→
   186→  const handleSubmit = async () => {
   187→    if (!title || !courseId || !venueId || !level || selectedDepts.length === 0 || !scheduledAt) {
   188→      toast.error('Please fill in all required fields');
   189→      return;
   190→    }
   191→
   192→    setSaving(true);
   193→    try {
   194→      const res = await fetch('/api/lecturer/sessions', {
   195→        method: 'POST',
   196→        headers: { 'Content-Type': 'application/json' },
   197→        body: JSON.stringify({
   198→          title,
   199→          courseId,
   200→          venueId,
   201→          lecturerId,
   202→          level: parseInt(level, 10),
   203→          departmentIds: selectedDepts,
   204→          distanceThreshold,
   205→          durationMinutes,
   206→          scheduledAt,
   207→        }),
   208→      });
   209→      const json = await res.json();
   210→
   211→      if (json.success) {
   212→        toast.success('Session created successfully');
   213→        resetForm();
   214→        onOpenChange(false);
   215→        onCreated();
   216→      } else if (res.status === 409) {
   217→        const conflicts = json.data?.conflicts || [];
   218→        if (json.error?.includes('Venue')) {
   219→          const conflictList = conflicts.map((c: { title: string; courseName: string; scheduledAt: string }) =>
   220→            `"${c.title}" (${c.courseName}) at ${format(new Date(c.scheduledAt), 'PPp')}`
   221→          ).join(', ');
   222→          toast.error(`Venue conflict: ${conflictList}`, { duration: 8000 });
   223→        } else if (json.error?.includes('department') || json.error?.includes('Department')) {
   224→          const conflictList = conflicts.map((c: { title: string; conflictingDepartments?: string[] }) =>
   225→            `"${c.title}" (Depts: ${c.conflictingDepartments?.join(', ')})`
   226→          ).join(', ');
   227→          toast.error(`Department conflict: ${conflictList}`, { duration: 8000 });
   228→        } else {
   229→          toast.error(json.error || 'Scheduling conflict detected');
   230→        }
   231→      } else {
   232→        toast.error(json.error || 'Failed to create session');
   233→      }
   234→    } catch {
   235→      toast.error('Network error creating session');
   236→    } finally {
   237→      setSaving(false);
   238→    }
   239→  };
   240→
   241→  // Filter departments for selected course - if course has specific departments, show those; otherwise show all
   242→  const selectedCourse = courses.find((c) => c.id === courseId);
   243→  const courseDepartments = selectedCourse?.departments?.length
   244→    ? selectedCourse.departments
   245→    : departments;
   246→
   247→  return (
   248→    <Dialog open={open} onOpenChange={onOpenChange} modal={false}>
   249→      <DialogContent className="max-w-2xl max-h-[85vh] overflow-y-auto">
   250→        <DialogHeader>
   251→          <DialogTitle>Create New Session</DialogTitle>
   252→        </DialogHeader>
   253→
   254→        <div className="space-y-4 py-4">
   255→          {/* Title */}
   256→          <div className="space-y-2">
   257→            <Label htmlFor="session-title">Session Title *</Label>
   258→            <Input
   259→              id="session-title"
   260→              value={title}
   261→              onChange={(e) => setTitle(e.target.value)}
   262→              placeholder="e.g. Week 5 Lecture"
   263→            />
   264→          </div>
   265→
   266→          {/* Course & Level */}
   267→          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   268→            <div className="space-y-2">
   269→              <Label>Course *</Label>
   270→              <select
   271→                className="flex h-9 w-full rounded-md border border-input bg-transparent px-3 py-1 text-sm shadow-sm transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring disabled:cursor-not-allowed disabled:opacity-50"
   272→                value={courseId}
   273→                onChange={(e) => {
   274→                  const val = e.target.value;
   275→                  setCourseId(val);
   276→                  setSelectedDepts([]);
   277→                  const course = courses.find((c) => c.id === val);
   278→                  if (course) setLevel(String(course.level ?? 100));
   279→                }}
   280→              >
   281→                <option value="">Select course</option>
   282→                {courses.length === 0 ? (
   283→                  <option value="" disabled>No courses assigned yet</option>
   284→                ) : (
   285→                  courses.map((c) => (
   286→                    <option key={c.id} value={c.id}>
   287→                      {c.code} - {c.name}
   288→                    </option>
   289→                  ))
   290→                )}
   291→              </select>
   292→            </div>
   293→
   294→            <div className="space-y-2">
   295→              <Label htmlFor="session-level">Level *</Label>
   296→              <Select value={level} onValueChange={setLevel}>
   297→                <SelectTrigger id="session-level" className="w-full">
   298→                  <SelectValue placeholder="Select level" />
   299→                </SelectTrigger>
   300→                <SelectContent>
   301→                  {VALID_LEVELS.map((l) => (
   302→                    <SelectItem key={String(l)} value={String(l)}>
   303→                      {l} Level
   304→                    </SelectItem>
   305→                  ))}
   306→                </SelectContent>
   307→              </Select>
   308→            </div>
   309→          </div>
   310→
   311→          {/* Venue */}
   312→          <div className="space-y-2">
   313→            <Label>Venue *</Label>
   314→            <select
   315→              className="flex h-9 w-full rounded-md border border-input bg-transparent px-3 py-1 text-sm shadow-sm transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring disabled:cursor-not-allowed disabled:opacity-50"
   316→              value={venueId}
   317→              onChange={(e) => setVenueId(e.target.value)}
   318→            >
   319→              <option value="">Select venue</option>
   320→              {venues.length === 0 ? (
   321→                <option value="" disabled>No venues available</option>
   322→              ) : (
   323→                venues.map((v) => (
   324→                  <option key={v.id} value={v.id}>
   325→                    {v.name}
   326→                  </option>
   327→                ))
   328→              )}
   329→            </select>
   330→          </div>
   331→
   332→          {/* Departments */}
   333→          <div className="space-y-2">
   334→            <Label>Departments * (select one or more)</Label>
   335→            <div className="grid grid-cols-2 md:grid-cols-3 gap-2 max-h-40 overflow-y-auto border rounded-md p-3 bg-blue-50/30 dark:bg-blue-950/10">
   336→              {courseDepartments.map((dept) => (
   337→                <div key={dept.id} className="flex items-center space-x-2">
   338→                  <Checkbox
   339→                    id={`dept-${dept.id}`}
   340→                    checked={selectedDepts.includes(dept.id)}
   341→                    onCheckedChange={() => handleDeptToggle(dept.id)}

   345→                    className="text-sm leading-none cursor-pointer"
   346→                  >
   347→                    {dept.name}
   348→                  </label>
   349→                </div>
   350→              ))}
   351→              {courseDepartments.length === 0 && (
   352→                <p className="text-xs text-muted-foreground col-span-3">
   353→                  No departments available. Ask admin to create departments.

   364→              onValueChange={([val]) => setDistanceThreshold(val)}
   365→              min={10}
   366→              max={500}
   367→              step={10}
   368→            />
   369→            <div className="flex justify-between text-xs text-muted-foreground">
   370→              <span>10m</span>

   378→            <Input
   379→              id="session-duration"
   380→              type="number"
   381→              min={5}
   382→              max={120}
   383→              value={durationMinutes}
   384→              onChange={(e) => setDurationMinutes(parseInt(e.target.value) || 15)}
   385→            />
   386→          </div>
   387→
   388→          {/* Scheduled Date/Time */}
   389→          <div className="space-y-2">
   390→            <Label htmlFor="session-scheduled">Scheduled Date & Time *</Label>
   391→            <Input
   392→              id="session-scheduled"

   407→                <Loader2 className="mr-2 h-4 w-4 animate-spin" /> Creating

   410→              <>
   411→                <Plus className="mr-2 h-4 w-4" /> Create Session

   413→            )}
   414→          </Button>
   415→        </DialogFooter>
   416→      </DialogContent>
   417→    </Dialog>
   418→  );
   419→}

   423→// ============================================================
   424→interface LiveMonitorProps {
   425→  session: SessionInfo;
   426→}
   427→
   428→function LiveMonitor({ session }: LiveMonitorProps) {
   429→  const [attendances, setAttendances] = useState<AttendanceInfo[]>([]);
   430→  const [loading, setLoading] = useState(true);
   431→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   432→
   433→  const fetchAttendances = useCallback(async () => {
   434→    try {
   435→      const res = await fetch(`/api/lecturer/analytics?sessionId=${session.id}`);
   436→      const json = await res.json();
   437→      if (json.success && json.data) {
   438→        setAttendances(json.data.attendances || []);
   439→      }
   440→    } catch {
   441→      // Silently fail on polling
   442→    } finally {
   443→      setLoading(false);
   444→    }
   445→  }, [session.id]);
   446→
   447→  useEffect(() => {
   448→    fetchAttendances();
   449→    intervalRef.current = setInterval(fetchAttendances, ATTENDANCE_POLL_INTERVAL);
   450→    return () => {
   451→      if (intervalRef.current) clearInterval(intervalRef.current);
   452→    };
   453→  }, [fetchAttendances]);
   454→
   455→  const present = attendances.filter((a) => a.status === 'present').length;
   456→  const pending = attendances.filter((a) => a.status === 'pending_review' || a.status === 'pending').length;
   457→  const rejected = attendances.filter(
   458→    (a) => a.status === 'rejected_location' || a.status === 'rejected_identity'
   459→  ).length;
   460→  const total = session.totalTargetStudents || 0;
   461→
   462→  return (
   463→    <Card className="card-elevated border-emerald-200 dark:border-emerald-800 bg-emerald-50/20 dark:bg-emerald-950/10">
   464→      <CardHeader className="pb-3">
   465→        <div className="flex items-center gap-2">
   466→          <Radio className="h-4 w-4 text-emerald-600 animate-pulse" />
   467→          <CardTitle className="text-base">Live Monitor</CardTitle>
   468→        </div>
   469→      </CardHeader>
   470→      <CardContent className="space-y-4">
   471→        {/* Stats */}
   472→        <div className="grid grid-cols-3 gap-3">
   473→          <div className="text-center p-3 rounded-lg bg-emerald-50 dark:bg-emerald-950/30">
   474→            <div className="text-2xl font-bold text-emerald-600 dark:text-emerald-400">{present}</div>
   475→            <div className="text-xs text-muted-foreground">Present</div>
   476→          </div>
   477→          <div className="text-center p-3 rounded-lg bg-amber-50 dark:bg-amber-950/30">
   478→            <div className="text-2xl font-bold text-amber-600 dark:text-amber-400">{pending}</div>
   479→            <div className="text-xs text-muted-foreground">Pending</div>
   480→          </div>
   481→          <div className="text-center p-3 rounded-lg bg-red-50 dark:bg-red-950/30">
   482→            <div className="text-2xl font-bold text-red-600 dark:text-red-400">{rejected}</div>
   483→            <div className="text-xs text-muted-foreground">Rejected</div>
   484→          </div>
   485→        </div>
   486→
   487→        {/* Progress */}
   488→        {total > 0 && (
   489→          <div className="space-y-1">
   490→            <div className="flex justify-between text-xs text-muted-foreground">
   491→              <span>Check-in Progress</span>
   492→              <span>{present + pending + rejected} / {total}</span>
   493→            </div>
   494→            <Progress value={((present + pending + rejected) / total) * 100} className="h-2" />
   495→          </div>
   496→        )}
   497→
   498→        {/* Recent Check-ins */}
   499→        <div className="max-h-60 overflow-y-auto custom-scrollbar">
   500→          {loading ? (
   501→            <div className="space-y-2">
   502→              {Array.from({ length: 3 }).map((_, i) => (
   503→                <Skeleton key={i} className="h-12" />
   504→              ))}
   505→            </div>
   506→          ) : attendances.length === 0 ? (
   507→            <p className="text-center text-sm text-muted-foreground py-4">
   508→              No check-ins yet
   509→            </p>
   510→          ) : (
   511→            <Table>
   512→              <TableHeader>
   513→                <TableRow>
   514→                  <TableHead>Student</TableHead>
   515→                  <TableHead>Time</TableHead>
   516→                  <TableHead>Score</TableHead>
   517→                  <TableHead>Status</TableHead>
   518→                </TableRow>
   519→              </TableHeader>
   520→              <TableBody>
   521→                {attendances
   522→                  .filter((a) => a.checkInTime)
   523→                  .sort((a, b) => new Date(b.checkInTime!).getTime() - new Date(a.checkInTime!).getTime())
   524→                  .map((a) => (
   525→                    <TableRow key={a.id}>
   526→                      <TableCell className="font-medium text-sm">
   527→                        {a.studentName || a.studentId.slice(0, 8)}
   528→                      </TableCell>
   529→                      <TableCell className="text-sm">
   530→                        {a.checkInTime ? format(new Date(a.checkInTime), 'HH:mm:ss') : '—'}
   531→                      </TableCell>
   532→                      <TableCell className="text-sm">
   533→                        {a.similarityScore !== null ? `${a.similarityScore.toFixed(1)}%` : '—'}
   534→                      </TableCell>
   535→                      <TableCell>
   536→                        <AttendanceStatusBadge status={a.status} />
   537→                      </TableCell>
   538→                    </TableRow>
   539→                  ))}
   540→              </TableBody>
   541→            </Table>
   542→          )}
   543→        </div>
   544→      </CardContent>
   545→    </Card>
   546→  );
   547→}

   784→function ReviewQueueTab({ lecturerId }: ReviewQueueTabProps) {
   785→  const [queue, setQueue] = useState<
   786→    {
   787→      id: string;

   835→        toast.error(json.error || 'Failed to process action');

   838→      toast.error('Network error');

   840→      setActioning(null);

   920→                        <Loader2 className="h-3.5 w-3.5 animate-spin" />
