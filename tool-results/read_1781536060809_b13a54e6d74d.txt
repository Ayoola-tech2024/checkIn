   108→function ActivationFlow() {
   109→  const { user, updateUser, logout } = useAuthStore();
   110→  const [step, setStep] = useState<ActivationStep>('email');
   111→  const [email, setEmail] = useState('');
   112→  const [password, setPassword] = useState('');
   113→  const [confirmPassword, setConfirmPassword] = useState('');
   114→  const [faceData, setFaceData] = useState<{ selfieData: string; facialDescriptor: number[] } | null>(null);
   115→  const [activating, setActivating] = useState(false);
   116→  const [errors, setErrors] = useState<Record<string, string>>({});
   117→
   118→  const validateEmailStep = useCallback(() => {
   119→    const newErrors: Record<string, string> = {};
   120→    if (!email.trim()) {
   121→      newErrors.email = 'Email is required';
   122→    } else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   123→      newErrors.email = 'Invalid email format';
   124→    }
   125→    if (!password) {
   126→      newErrors.password = 'Password is required';
   127→    } else if (password.length < 6) {
   128→      newErrors.password = 'Password must be at least 6 characters';
   129→    }
   130→    if (password !== confirmPassword) {
   131→      newErrors.confirmPassword = 'Passwords do not match';
   132→    }
   133→    setErrors(newErrors);
   134→    return Object.keys(newErrors).length === 0;
   135→  }, [email, password, confirmPassword]);
   136→
   137→  const handleEmailSubmit = useCallback(
   138→    (e: React.FormEvent) => {
   139→      e.preventDefault();
   140→      if (validateEmailStep()) {
   141→        setStep('selfie');
   142→      }
   143→    },
   144→    [validateEmailStep]
   145→  );
   146→
   147→  const handleFaceCapture = useCallback(
   148→    (data: { selfieData: string; facialDescriptor: number[] }) => {
   149→      setFaceData(data);
   150→      setStep('processing');
   151→
   152→      // Call activate API
   153→      activateAccount(data);
   154→    },
   155→    []
   156→  );
   157→
   158→  const activateAccount = useCallback(
   159→    async (data: { selfieData: string; facialDescriptor: number[] }) => {
   160→      if (!user?.id) return;
   161→
   162→      setActivating(true);
   163→
   164→      try {
   165→        const response = await fetch('/api/student/activate', {
   166→          method: 'POST',
   167→          headers: { 'Content-Type': 'application/json' },
   168→          body: JSON.stringify({
   169→            studentId: user.id,
   170→            email,
   171→            password,
   172→            facialData: JSON.stringify({ descriptor: data.facialDescriptor }),
   173→            selfieData: data.selfieData,
   174→          }),
   175→        });
   176→
   177→        const result: ApiResponse = await response.json();
   178→
   179→        if (result.success && result.data) {
   180→          setStep('complete');
   181→          updateUser({
   182→            activated: true,
   183→            email,
   184→          });
   185→          toast.success('Account activated successfully!');
   186→        } else {
   187→          toast.error(result.error || 'Activation failed');
   188→          setStep('selfie');
   189→          setFaceData(null);
   190→        }
   191→      } catch {
   192→        toast.error('Network error. Please try again.');
   193→        setStep('selfie');
   194→        setFaceData(null);
   195→      } finally {
   196→        setActivating(false);
   197→      }
   198→    },
   199→    [user?.id, email, password, updateUser]
   200→  );
   201→
   202→  const handleFaceError = useCallback((error: string) => {
   203→    toast.error(error);
   204→  }, []);
   205→
   206→  // Step indicator
   207→  const steps = [
   208→    { key: 'email', label: 'Link Email', icon: Mail },
   209→    { key: 'selfie', label: 'Capture Selfie', icon: Camera },
   210→    { key: 'complete', label: 'Complete', icon: CheckCircle2 },
   211→  ] as const;
   212→
   213→  const currentStepIndex = steps.findIndex(
   214→    (s) => s.key === (step === 'processing' ? 'selfie' : step)
   215→  );
   216→
   217→  return (
   218→    <div className="min-h-screen flex flex-col items-center justify-center p-4 sm:p-6 bg-page-gradient">
   219→      <div className="w-full max-w-md space-y-6">
   220→        {/* Header with back to login */}
   221→        <div className="text-center space-y-2">
   222→          <div className="flex items-center justify-center gap-2">
   223→            <div className="flex items-center justify-center size-10 rounded-xl header-gradient text-white shadow-sm">
   224→              <GraduationCap className="h-5 w-5" />
   225→            </div>
   226→            <h1 className="text-2xl font-bold text-primary">checkIn</h1>
   227→          </div>
   228→          <p className="text-sm text-muted-foreground">Activate your student account</p>
   229→          <button
   230→            onClick={logout}
   231→            className="inline-flex items-center gap-1.5 text-xs text-muted-foreground hover:text-foreground transition-colors mt-1"
   232→          >
   233→            <ArrowLeft className="h-3 w-3" />
   234→            Back to Login
   235→          </button>
   236→        </div>
   237→
   238→        {/* Step indicators */}
   239→        <div className="flex items-center justify-center gap-2">
   240→          {steps.map((s, i) => {
   241→            const Icon = s.icon;
   242→            const isActive = i === currentStepIndex;
   243→            const isDone = i < currentStepIndex;
   244→            return (
   245→              <div key={s.key} className="flex items-center gap-2">
   246→                <div
   247→                  className={`flex items-center gap-1.5 px-3 py-1.5 rounded-full text-xs font-medium transition-colors ${
   248→                    isActive
   249→                      ? 'bg-primary text-primary-foreground shadow-sm'
   250→                      : isDone
   251→                        ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-400'
   252→                        : 'bg-muted text-muted-foreground'
   253→                  }`}
   254→                >
   255→                  <Icon className="h-3.5 w-3.5" />
   256→                  <span className="hidden sm:inline">{s.label}</span>
   257→                </div>
   258→                {i < steps.length - 1 && (
   259→                  <ChevronRight className="h-4 w-4 text-muted-foreground" />
   260→                )}
   261→              </div>
   262→            );
   263→          })}
   264→        </div>
   265→
   266→        {/* Step 1: Email & Password */}
   267→        {step === 'email' && (
   268→          <Card className="card-elevated border-0 shadow-sm">
   269→            <CardHeader className="pb-3">
   270→              <CardTitle className="text-lg flex items-center gap-2">
   271→                <Mail className="h-5 w-5 text-primary" />
   272→                Link Your Email
   273→              </CardTitle>
   274→            </CardHeader>
   275→            <CardContent>
   276→              <form onSubmit={handleEmailSubmit} className="space-y-4">
   277→                <div className="space-y-2">
   278→                  <Label htmlFor="email">Email Address</Label>
   279→                  <Input
   280→                    id="email"
   281→                    type="email"
   282→                    placeholder="your.email@university.edu"
   283→                    value={email}
   284→                    onChange={(e) => setEmail(e.target.value)}
   285→                    className={errors.email ? 'border-red-500' : ''}
   286→                    autoFocus
   287→                  />
   288→                  {errors.email && (
   289→                    <p className="text-xs text-red-500">{errors.email}</p>
   290→                  )}
   291→                </div>
   292→                <div className="space-y-2">
   293→                  <Label htmlFor="password">Create Password</Label>
   294→                  <Input
   295→                    id="password"
   296→                    type="password"
   297→                    placeholder="At least 6 characters"
   298→                    value={password}
   299→                    onChange={(e) => setPassword(e.target.value)}
   300→                    className={errors.password ? 'border-red-500' : ''}
   301→                  />
   302→                  {errors.password && (
   303→                    <p className="text-xs text-red-500">{errors.password}</p>
   304→                  )}
   305→                </div>
   306→                <div className="space-y-2">
   307→                  <Label htmlFor="confirmPassword">Confirm Password</Label>
   308→                  <Input
   309→                    id="confirmPassword"
   310→                    type="password"
   311→                    placeholder="Confirm your password"
   312→                    value={confirmPassword}
   313→                    onChange={(e) => setConfirmPassword(e.target.value)}
   314→                    className={errors.confirmPassword ? 'border-red-500' : ''}
   315→                  />
   316→                  {errors.confirmPassword && (
   317→                    <p className="text-xs text-red-500">{errors.confirmPassword}</p>
   318→                  )}
   319→                </div>
   320→                <Button type="submit" className="w-full gap-2">
   321→                  Continue
   322→                  <ArrowRight className="h-4 w-4" />
   323→                </Button>
   324→              </form>
   325→            </CardContent>
   326→          </Card>
   327→        )}
   328→
   329→        {/* Step 2: Selfie Capture */}
   330→        {step === 'selfie' && (
   331→          <div className="space-y-4">
   332→            <Card className="card-elevated border-0 shadow-sm">
   333→              <CardHeader className="pb-3">
   334→                <div className="flex items-center justify-between">
   335→                  <CardTitle className="text-lg flex items-center gap-2">
   336→                    <Camera className="h-5 w-5 text-primary" />
   337→                    Capture Your Selfie
   338→                  </CardTitle>
   339→                  <Button
   340→                    variant="ghost"
   341→                    size="sm"
   342→                    onClick={() => { setStep('email'); setFaceData(null); }}
   343→                    className="gap-1.5 text-muted-foreground"
   344→                  >
   345→                    <ArrowLeft className="h-4 w-4" />
   346→                    Back
   347→                  </Button>
   348→                </div>
   349→              </CardHeader>
   350→              <CardContent>
   351→                <div className="rounded-lg bg-blue-50/50 dark:bg-blue-950/20 p-3">
   352→                  <p className="text-sm text-muted-foreground">
   353→                    This photo will be used for identity verification during attendance check-in.
   354→                    Make sure your face is clearly visible and well-lit.
   355→                  </p>
   356→                </div>
   357→              </CardContent>
   358→            </Card>
   359→            <FaceCapture
   360→              onCapture={handleFaceCapture}
   361→              mode="activation"
   362→              onError={handleFaceError}
   363→            />
   364→          </div>
   365→        )}
   366→
   367→        {/* Step 3: Processing */}
   368→        {step === 'processing' && (
   369→          <Card className="card-elevated border-0 shadow-sm">
   370→            <CardContent className="p-6">
   371→              <div className="flex flex-col items-center justify-center gap-4 py-8">
   372→                <div className="relative">
   373→                  <div className="h-14 w-14 rounded-full bg-primary/10 flex items-center justify-center">
   374→                    <Loader2 className="h-8 w-8 animate-spin text-primary" />
   375→                  </div>
   376→                </div>
   377→                <div className="text-center">
   378→                  <p className="font-semibold">Activating Your Account</p>
   379→                  <p className="text-sm text-muted-foreground mt-1">
   380→                    Processing your facial data and setting up your profile...
   381→                  </p>
   382→                </div>
   383→              </div>
   384→            </CardContent>
   385→          </Card>
   386→        )}
   387→
   388→        {/* Step 4: Complete */}
   389→        {step === 'complete' && (
   390→          <Card className="card-elevated border-0 shadow-sm">
   391→            <CardContent className="p-6">
   392→              <div className="flex flex-col items-center justify-center gap-4 py-8">
   393→                <div className="h-16 w-16 rounded-full bg-emerald-100 dark:bg-emerald-950 flex items-center justify-center ring-4 ring-emerald-100/50 dark:ring-emerald-950/50">
   394→                  <CheckCircle2 className="h-10 w-10 text-emerald-500" />
   395→                </div>
   396→                <div className="text-center">
   397→                  <p className="text-xl font-bold text-emerald-600 dark:text-emerald-400">
   398→                    Account Activated!
   399→                  </p>
   400→                  <p className="text-sm text-muted-foreground mt-2">
   401→                    Your account is ready. You can now check in to attendance sessions.
   402→                  </p>
   403→                </div>
   404→              </div>
   405→            </CardContent>
   406→          </Card>
   407→        )}
   408→      </div>
   409→    </div>
   410→  );
   411→}

   417→function ActivePortal() {
   418→  const { user, logout } = useAuthStore();
   419→  const [sessions, setSessions] = useState<SessionWithAttendance[]>([]);
   420→  const [loading, setLoading] = useState(true);
   421→  const [activeCheckIn, setActiveCheckIn] = useState<SessionInfo | null>(null);
   422→  const [profileOpen, setProfileOpen] = useState(false);
   423→  const [stats, setStats] = useState<StudentStats | null>(null);
   424→  const [statsLoading, setStatsLoading] = useState(true);
   425→  const pollRef = useRef<ReturnType<typeof setInterval> | null>(null);
   426→
   427→  // Fetch sessions
   428→  const fetchSessions = useCallback(async () => {
   429→    if (!user?.id) return;
   430→
   431→    try {
   432→      const response = await fetch(`/api/student/sessions?studentId=${user.id}`);
   433→      const result: ApiResponse<SessionWithAttendance[]> = await response.json();
   434→
   435→      if (result.success && result.data) {
   436→        setSessions(result.data);
   437→      }
   438→    } catch {
   439→      // Silently fail for polling - don't spam errors
   440→    } finally {
   441→      setLoading(false);
   442→    }
   443→  }, [user?.id]);
   444→
   445→  // Fetch stats
   446→  const fetchStats = useCallback(async () => {
   447→    if (!user?.id) return;
   448→
   449→    try {
   450→      const response = await fetch(`/api/student/stats?studentId=${user.id}`);
   451→      const result: ApiResponse<StudentStats> = await response.json();
   452→
   453→      if (result.success && result.data) {
   454→        setStats(result.data);
   455→      }
   456→    } catch {
   457→      // Silently fail - stats are not critical
   458→    } finally {
   459→      setStatsLoading(false);
   460→    }
   461→  }, [user?.id]);
   462→
   463→  // Initial fetch + polling
   464→  useEffect(() => {
   465→    fetchSessions();
   466→    fetchStats();
   467→
   468→    pollRef.current = setInterval(fetchSessions, SESSION_POLL_INTERVAL);
   469→
   470→    return () => {
   471→      if (pollRef.current) {
   472→        clearInterval(pollRef.current);
   473→      }
   474→    };
   475→  }, [fetchSessions, fetchStats]);
   476→
   477→  // Handle check-in completion
   478→  const handleCheckInComplete = useCallback(
   479→    (result: CheckInResult) => {
   480→      setActiveCheckIn(null);
   481→      // Refresh sessions and stats to get updated attendance status
   482→      fetchSessions();
   483→      fetchStats();
   484→
   485→      if (result.success) {
   486→        toast.success(result.message);
   487→      } else {
   488→        toast.error(result.message);
   489→      }
   490→    },
   491→    [fetchSessions, fetchStats]
   492→  );
   493→
   494→  const handleLogout = useCallback(() => {
   495→    logout();
   496→    toast.info('Logged out successfully');
   497→  }, [logout]);
   498→
   499→  // Categorize sessions
   500→  const activeSessions = sessions.filter((s) => s.status === 'active');
   501→  const upcomingSessions = sessions.filter((s) => s.status === 'scheduled');
   502→  const pastSessions = sessions.filter(
   503→    (s) => s.status === 'completed' || s.status === 'cancelled'
   504→  );
   505→
   506→  // Check-in flow is active
   507→  if (activeCheckIn) {
   508→    return (
   509→      <div className="min-h-screen p-4 sm:p-6 bg-page-gradient">
   510→        <div className="max-w-lg mx-auto pt-4">
   511→          <CheckInFlow
   512→            session={activeCheckIn}
   513→            studentId={user!.id}
   514→            onComplete={handleCheckInComplete}
   515→            onCancel={() => setActiveCheckIn(null)}
   516→          />
   517→        </div>
   518→      </div>
   519→    );
   520→  }
   521→
   522→  return (
   523→    <div className="min-h-screen flex flex-col bg-page-gradient">
   524→      {/* Header */}
   525→      <header className="sticky top-0 z-10 bg-card/80 backdrop-blur-sm border-b border-border/60">
   526→        <div className="header-gradient">
   527→          <div className="max-w-2xl mx-auto px-4 py-3 flex items-center justify-between">
   528→            <div className="flex items-center gap-2 min-w-0">
   529→              <div className="flex items-center justify-center size-8 rounded-lg bg-white/20 shrink-0">
   530→                <GraduationCap className="h-4 w-4 text-white" />
   531→              </div>
   532→              <div className="min-w-0">
   533→                <h1 className="text-sm font-semibold truncate text-white">{user?.name}</h1>
   534→                <p className="text-xs text-white/70 truncate flex items-center gap-1.5">
   535→                  {user?.matricNumber} • {user?.departmentName} • Level {user?.level ?? '—'}
   536→                  <Badge className="bg-white/20 text-white border-white/30 text-[9px] px-1 py-0 hover:bg-white/30">{SCHOOL}</Badge>
   537→                </p>
   538→              </div>
   539→            </div>
   540→            <div className="flex items-center gap-2 shrink-0">
   541→              <Button
   542→                variant="ghost"
   543→                size="icon"
   544→                onClick={() => setProfileOpen(true)}
   545→                className="h-9 w-9 text-white/80 hover:text-white hover:bg-white/10"
   546→                title="View profile"
   547→              >
   548→                <UserCircle className="h-4 w-4" />
   549→              </Button>
   550→              <Button
   551→                variant="ghost"
   552→                size="icon"
   553→                onClick={fetchSessions}
   554→                className="h-9 w-9 text-white/80 hover:text-white hover:bg-white/10"
   555→                title="Refresh sessions"
   556→              >
   557→                <RefreshCw className="h-4 w-4" />
   558→              </Button>
   559→              <Button
   560→                variant="ghost"
   561→                size="icon"
   562→                onClick={handleLogout}
   563→                className="h-9 w-9 text-white/80 hover:text-red-200 hover:bg-white/10"
   564→                title="Log out"
   565→              >
   566→                <LogOut className="h-4 w-4" />
   567→              </Button>
   568→            </div>
   569→          </div>
   570→        </div>
   571→      </header>
   572→
   573→      {/* Main Content */}
   574→      <main className="flex-1 max-w-2xl mx-auto w-full px-4 py-4 space-y-6">
   575→        {/* Loading State */}
   576→        {loading && (
   577→          <div className="space-y-4">
   578→            <div className="space-y-2">
   579→              <Skeleton className="h-5 w-32" />
   580→              <Skeleton className="h-40 w-full rounded-xl" />
   581→            </div>
   582→            <div className="space-y-2">
   583→              <Skeleton className="h-5 w-40" />
   584→              <Skeleton className="h-32 w-full rounded-xl" />
   585→            </div>
   586→          </div>
   587→        )}
   588→
   589→        {!loading && sessions.length === 0 && stats && stats.totalSessions === 0 && (
   590→          <div className="text-center py-16">
   591→            <div className="inline-flex items-center justify-center size-16 rounded-2xl bg-blue-50/80 dark:bg-blue-950/30 mb-4">
   592→              <BookOpen className="h-8 w-8 text-primary/40" />
   593→            </div>
